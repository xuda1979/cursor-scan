def main(argv: List[str] | None = None) -> int:
    args = _arg_parser().parse_args(argv)
    options = ScanOptions(max_bytes=args.max_bytes,
                          ignore_patterns=compile_ignores(args.ignore))

    if args.history:
        return _cmd_history(args.history_dir)
    if args.serve:
        return _cmd_serve(args.history_dir, args.host, args.port)
    if args.diff:
        try:
            before = load_snapshot(resolve_snapshot(args.diff[0], args.history_dir))
            after = load_snapshot(resolve_snapshot(args.diff[1], args.history_dir))
        except FileNotFoundError as exc:
            print(f"cursor-scan: {exc}", file=sys.stderr)
            return 2
        print_diff(diff_snapshots(before, after))

        print(f"cursor-scan: 无权限读取目标: {exc}", file=sys.stderr)
        return 2

    saved = None
    if not args.no_save:
        try:
            saved = save_snapshot(result, args.history_dir)
        except OSError as exc:
            print(f"cursor-scan: 警告: 保存历史快照失败: {exc}", file=sys.stderr)

    if args.format == "json":
        print(json.dumps(result_to_dict(result), ensure_ascii=False, indent=2))
    elif args.format == "sarif":
        print(json.dumps(result_to_sarif(result), ensure_ascii=False, indent=2))

        print_text_report(result)

def print_diff(diff) -> None:
    """Render a two-snapshot ScanDiff as added/fixed/kept finding lines."""
    delta = diff.score_delta
    sign = "+" if delta > 0 else ""
    print(f"评分: {diff.before.get('score', '?')} -> {diff.after.get('score', '?')} "
          f"({sign}{delta})  等级: {diff.before.get('grade', '?')} -> "
          f"{diff.after.get('grade', '?')}")
    print(f"新增 {len(diff.added)} 项, 修复 {len(diff.fixed)} 项, "
          f"仍在 {len(diff.kept)} 项")
    if diff.suppressed:
        print(f"忽略 {len(diff.suppressed)} 项 (后一次扫描中被 --ignore 抑制)\n")
    for label, items in (("新增", diff.added), ("修复", diff.fixed),

def _sarif_rules(findings: List[Evidence]) -> List[dict]:
    seen: Dict[str, dict] = {}
    for f in findings:
        if f.rule_id in seen:
            continue
        seen[f.rule_id] = {
            "id": f.rule_id,
            "name": f.rule_id,
            "shortDescription": {"text": f.title},
            "help": {"text": f.fix, "markdown": f.fix},
            "defaultConfiguration": {"level": "error"},
            "properties": {"dimension": f.dimension},
        }
    return list(seen.values())

def slug_for_target(target: str) -> str:
    base = os.path.basename(os.path.normpath(target)) or "root"
    slug = re.sub(r"[^A-Za-z0-9._-]+", "-", base).strip("-").lower()
    return slug or "project"


def snapshot_path(target: str, history_dir: str,
                  now: Optional[datetime] = None) -> str:
    slug = slug_for_target(target)
    ts = (now or datetime.now(timezone.utc)).strftime(_TIMESTAMP_FMT)
    base = os.path.join(history_dir, f"{slug}-{ts}")
    path = base + ".json"
    i = 2
    while os.path.exists(path):  # two scans in the same second
        path = f"{base}-{i}.json"
        i += 1
    return path

def save_snapshot(result: ScanResult, history_dir: Optional[str] = None) -> str:
    """Persist a ScanResult as JSON and return the written path.

    Creates the history directory if needed and never raises on a clean
    run; callers that care about durability should check ``os.path.exists``.
    """
    hd = history_dir or default_history_dir()
    os.makedirs(hd, exist_ok=True)
    path = snapshot_path(result.target, hd)
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(result_to_dict(result), fh, ensure_ascii=False, indent=2)
    return path

def resolve_snapshot(ref: str, history_dir: Optional[str] = None) -> str:
    """Turn a user reference into a snapshot path.

    ``ref`` is used verbatim when it names an existing file, otherwise it is
    treated as a target slug (or basename prefix) and resolved to the newest
    matching snapshot. Raises FileNotFoundError when nothing matches.
    """
    if os.path.isfile(ref):
        return ref
    hd = history_dir or default_history_dir()
    if not os.path.isdir(hd):
        raise FileNotFoundError(f"历史目录不存在: {hd}")
    slug = slug_for_target(ref)
    matches = [e.path for e in list_history(hd)
               if os.path.basename(e.path).startswith((slug, ref))]
    if not matches:
        raise FileNotFoundError(f"未找到快照: {ref} (历史目录: {hd})")
    return matches[0]  # list_history is newest-first


def load_snapshot(path: str) -> dict:

def _finding_key(f: dict) -> Tuple[str, str, int]:
    return (f.get("rule_id", "?"), os.path.normpath(f.get("file", "?")),
            f.get("line", 0))


def diff_snapshots(before: dict, after: dict) -> ScanDiff:
    """Compare two scan snapshots by (rule_id, file, line).

    Findings that existed in the before scan but only *disappear* because the
    after scan suppressed them (--ignore / .cursor-scan.json) are reported in
    ``suppressed`` rather than ``fixed``, so a diff never implies a fix that
    was actually just an ignore.
    """
    before_keyed = {_finding_key(f): f for f in before.get("findings", [])}
    after_keyed = {_finding_key(f): f for f in after.get("findings", [])}
    after_suppressed = {_finding_key(f): f
                        for f in after.get("suppressed", [])}
    before_suppressed = {_finding_key(f): f
                         for f in before.get("suppressed", [])}
    # A finding gone from after.findings is "suppressed" when it now sits in
    # after.suppressed (and is not itself a fix that re-appeared as active).
    suppressed = [f for k, f in sorted(before_keyed.items())
                  if k in after_suppressed and k not in after_keyed]
    fixed = [f for k, f in sorted(before_keyed.items())
             if k not in after_keyed and k not in after_suppressed]
    # A finding the after scan un-suppresses comes back active: report it as
    # "unsuppressed", not as a brand-new "added" instance.
    unsuppressed = [f for k, f in sorted(before_suppressed.items())
                    if k in after_keyed and k not in before_keyed]
    unsup_keys = {_finding_key(f) for f in unsuppressed}
    added = [f for k, f in sorted(after_keyed.items())
             if k not in before_keyed and k not in unsup_keys]
    kept = [f for k, f in sorted(after_keyed.items()) if k in before_keyed]
    return ScanDiff(before=before, after=after, added=added, fixed=fixed,
                    suppressed=suppressed, kept=kept,
                    unsuppressed=unsuppressed)

@dataclass
class IgnorePattern:
    """One parsed suppression pattern (all fields are optional)."""

    rule: Optional[str] = None
    file: Optional[str] = None
    line: Optional[int] = None

    @staticmethod
    def parse(spec: str) -> "IgnorePattern":
        parts = spec.split(":")
        while len(parts) < 3:
            parts.append(None)
        rule, file_pat, line_str = parts[0], parts[1], parts[2]
        # Empty fields from trailing/empty colon parts behave like omitted ones.
        rule = rule or None
        file_pat = file_pat or None
        line = int(line_str) if line_str and line_str.strip().isdigit() else None
        return IgnorePattern(rule=rule, file=file_pat, line=line)

    @staticmethod
    def _match(pat: Optional[str], *values: str) -> bool:
        return pat is None or any(fnmatchcase(v, pat) for v in values)

def compile_ignores(specs: List[str]) -> List[IgnorePattern]:
    """Parse a list of pattern strings into matchable ``IgnorePattern``s."""
    return [IgnorePattern.parse(s) for s in specs]

def analyze_code(source_path: str, code: str) -> List[Evidence]:
    """Regex fallback for weak-crypto findings."""
    findings: List[Evidence] = []

    def hit(line: int, rule_id: str, title: str, message: str, fix: str, sev: str = "high") -> Evidence:
        return Evidence(
            rule_id=rule_id,
            title=title,
            severity=sev,
            file=source_path,
            line=line,
            message=message,
            fix=fix,
            dimension="crypto",
            code_snippet="",
        )

    for lineno, raw in code_lines(code):
        line = raw
        # Import / from-import lines are nothing but a name binding; real usage
        # of a weak algorithm is detected at call sites (regex + AST).
        if line.lstrip().startswith(("import ", "from ")):
            continue
        if RE_HARDCODED_SEED.search(line):
            findings.append(hit(lineno, "crypto.hardcoded_seed", "硬编码随机种子",
                                "随机数种子固定使输出可预测", "移除固定 seed 或从环境变量读取"))
        if RE_HARDCODED_IV.search(line):
            findings.append(hit(lineno, "crypto.fixed_iv", "固定 IV/nonce",
                                "加密使用固定初始化向量导致相同明文产生相同密文", "每次加密使用随机 IV"))
        if RE_MD5_CALL.search(line):
            findings.append(hit(lineno, "crypto.md5_or_sha1", "弱哈希算法 (MD5)",
                                "MD5 已被破解，不应用于安全场景", "改用 SHA-256 / SHA-3 (hashlib.sha256)"))
        if RE_SHA1_HASH.search(line):
            findings.append(hit(lineno, "crypto.md5_or_sha1", "弱哈希算法 (SHA1)",
                                "SHA-1 碰撞攻击成本已被学界证实", "改用 SHA-256 / SHA-3"))
        if RE_DES.search(line):
            findings.append(hit(lineno, "crypto.weak_cipher_des", "弱加密算法 (DES)",
                                "DES 密钥长度过短，易被暴力破解", "改用 AES (GCM 模式)"))
        if RE_ECB.search(line):
            findings.append(hit(lineno, "crypto.ecb_mode", "不安全加密模式 (ECB)",
                                "ECB 模式相同明文块产生相同密文块，泄露结构信息", "改用 GCM / CBC + 随机 IV"))

def _ver_tuple(version: str) -> Optional[Tuple[int, ...]]:
    """Leading dotted-numeric prefix as an int tuple, else None."""
    m = re.match(r"^\s*v?(\d+(?:\.\d+)*)", str(version))
    if not m:
        return None
    return tuple(int(p) for p in m.group(1).split("."))

def _parse_pin(spec: str) -> Tuple[Optional[str], bool]:
    """Return (version, exact) for a requirements spec; exact==True only for ==."""
    m = _REQ_RE.match(spec)
    if not m:
        return None, False
    _, op, version = m.groups()
    return version, op in ("==", "===")


def _check_requirements(path: str, code: str, findings: List[Evidence]) -> None:
    for lineno, raw in enumerate(code.splitlines(), start=1):
        if not raw.strip() or _REQ_SKIP.match(raw):
            continue
        m = _REQ_RE.match(raw)
        if not m:
            continue
        name, op, version = m.groups()
        if not version or op not in ("==", "==="):
            findings.append(_unpinned_evidence(path, lineno, name, raw.strip()))
            continue
        vuln = _known_vuln(name, version)
        if vuln:
            findings.append(_vuln_evidence(path, lineno, name, f"=={version}", *vuln))

_FLOATING = ("*", "latest", "next", "x")


def _is_floating(spec: str) -> bool:
    s = spec.strip()

def _exact_npm_version(spec: str) -> Optional[str]:
    """Exact pins only (bare or ``=``); ranges like ^/~ never qualify."""
    s = spec.strip()
    if s.startswith(("^", "~", ">", "<")) or s in _FLOATING or s.endswith(".x"):
        return None
    m = re.match(r"=?v?(\d+(?:\.\d+)*)", s)
    if not m:
        return None
    return m.group(1)

def _check_package_json(path: str, code: str, findings: List[Evidence]) -> None:
    try:
        doc = json.loads(code)
    except ValueError:
        return  # malformed manifest: not part of this rule's scope
    if not isinstance(doc, dict):
        return
    source_lines = code.splitlines()
    for section in ("dependencies", "devDependencies"):
        deps = doc.get(section)
        if not isinstance(deps, dict):
            continue
        for name, spec in deps.items():
            if not isinstance(spec, str):
                continue
            lineno = _find_dep_line(source_lines, name)
            version = _exact_npm_version(spec)
            if version is None:
                if _is_floating(spec):
                    findings.append(_unpinned_evidence(path, lineno, name, spec))
                continue
            vuln = _known_vuln(name, version)
            if vuln:
                findings.append(_vuln_evidence(path, lineno, name, f"@{spec}", *vuln))

def _iter_source_files(root: str, opts: ScanOptions) -> Iterable[str]:
    """Yield relative paths of files worth scanning, in stable order."""
    root = os.path.abspath(root)
    for dirpath, dirnames, filenames in os.walk(root, followlinks=opts.follow_symlinks):
        dirnames[:] = sorted(
            d for d in dirnames
            if d not in SKIP_DIRS and not d.startswith(".")
        )
        for name in sorted(filenames):
            if name == CONFIG_NAME:
                continue  # the tool's own config is never scanned
            rel = os.path.relpath(os.path.join(dirpath, name), root)
            low = name.lower()
            ext = os.path.splitext(low)[1]
            if ext in SKIP_EXTS:
                continue
            if low in ALWAYS_TEXT or ext in TEXT_EXTS:
                yield rel


def _scan_file(root: str, rel: str, src: str, findings: List[Evidence]) -> None:
    if not src:
        return

    lang = ast_scanners.lang_for_file(rel)
    if lang == "python":
        try:
            ast_scanners.analyze_python(path_of(root, rel), src, findings)
        except Exception as exc:  # never let one file crash a scan
            findings.append(_internal_error(path_of(root, rel), exc))
    elif lang in ("javascript", "typescript"):
        try:
            ast_scanners.analyze_jsts(path_of(root, rel), src, findings)
        except Exception as exc:
            findings.append(_internal_error(path_of(root, rel), exc))

    # Regex fallback + secret rules apply to every text file; dependency-
    # manifest rules (requirements*.txt / package.json) are per-file too.
    for runner in (crypto, injection, deserialization, permissions):
        findings.extend(runner.analyze_code(path_of(root, rel), src))
    findings.extend(secrets.analyze_code(path_of(root, rel), src))
    findings.extend(deps.analyze_code(path_of(root, rel), src))
    findings.extend(xss.analyze_code(path_of(root, rel), src))

def _load_config(target: str) -> List[IgnorePattern]:
    """Load ``.cursor-scan.json`` from the target; return ignore patterns.

    A missing or malformed config is ignored (a warning would spam every scan
    on projects that never heard of this file).
    """
    cfg = os.path.join(target, CONFIG_NAME)
    if not os.path.isfile(cfg):
        return []
    try:
        with open(cfg, "r", encoding="utf-8") as fh:
            doc = json.load(fh)
    except (OSError, ValueError):
        return []
    specs = doc.get("ignore", []) if isinstance(doc, dict) else []
    if not isinstance(specs, list):
        return []
    return compile_ignores([s for s in specs if isinstance(s, str)])

def _backfill_snippets(result: ScanResult,
                       file_texts: Dict[str, List[Tuple[int, str]]],
                       target: str) -> None:
    """Resolve the offending source line for every finding.

    Rules report file/line only; the exact code fragment is filled in here so
    JSON / SARIF output and the web dashboard all carry the same snippet.
    ``access_control`` findings reference files by relative path while per-file
    rules use the absolute path, so both keys are indexed.
    """
    by_path: Dict[str, List[Tuple[int, str]]] = dict(file_texts)
    for rel, lines in file_texts.items():
        by_path[os.path.join(target, rel)] = lines
    for f in list(result.findings) + list(result.suppressed):
        if f.code_snippet or f.line <= 0:
            continue
        lines = by_path.get(f.file)
        if lines is None:
            try:
                lines = by_path.get(os.path.relpath(f.file, target))
            except ValueError:
                lines = None
        if lines and 0 < f.line <= len(lines):
            f.code_snippet = lines[f.line - 1][1].strip()

def collect_calls(root, src: str):
    """Yield (call_node) for every call node in the tree.

    Python grammar names calls ``call``; the JS/TS grammars use
    ``call_expression``.
    """
    stack = [root]
    while stack:
        node = stack.pop()
        if node.type in ("call", "call_expression"):
            yield node
        stack.extend(node.children)


def collect_new_expressions(root):
    """Yield every ``new_expression`` node (JS ``new Foo(...)``).

    tree-sitter models ``new Function(body)`` as a ``new_expression`` with a
    ``constructor`` field, never as a ``call_expression``, so ``collect_calls``
    cannot see it -- ``Function`` is the constructor-equivalent of ``eval`` and
    must be inspected on its own pass.
    """
    stack = [root]
    while stack:
        node = stack.pop()
        if node.type == "new_expression":
            yield node
        stack.extend(node.children)


def _bstr(bsrc: bytes, start: int, end: int) -> str:
    """Slice a node's source span and decode back to text.

    original ``str`` with those offsets corrupts every span after the first
    non-ASCII character, so spans are taken from the encoded bytes and decoded.

def _py_static_string(txt: str) -> str | None:
    """Value of a static quoted Python string literal; ``None`` otherwise.

    because they are unverifiable statically (dynamic f-strings / bytes hold

    if m.group(1):

    return m.group(3)

def _has_fstring_interp(args: str) -> bool:
    """True if ``args`` contains a Python f-string with ``{...}`` interpolation.

    Escaped literal braces (``{{ }}``) do not count as interpolation, so a
    constant f-string is left alone.  The scan is line-narrowed to the f-string
    body via ``re.S`` so embedded quotes/format specs inside the braces do not
    cut the match short.
    """
    for m in re.finditer(r'f(["\'])(.*?)\1', args, re.S):
        body = re.sub(r'\{\{.*?\}\}', "", m.group(2))
        if "{" in body:
            return True
    return False


def _has_concat(args: str) -> bool:
    if " + " in args or "+" in args or "%s" in args and " % " in args:
        return True
    # F-string / template-literal interpolation is the most common SQL
    # injection idiom in AI-generated code (e.g. ``execute(f"...{x}")``).
    if _has_fstring_interp(args):
        return True
    # str.format() interpolation: ``"SELECT ... {} ...".format(uid)`` -- the
    # format-spec braces plus a non-empty .format(...) argument list mark a
    # dynamic interpolation, so an inert ``"".format()`` stays quiet.
    if r".format(" in args and re.search(r"\.format\(\s*[^)\s]", args):
        return True
    if "`" in args and "${" in args:
        return True
    return False

_JS_PLAIN_STRING = re.compile(r"^(['\"])(.*?)\1$", re.S)

# Shell-interpreting binaries handed to subprocess/spawn-style calls.  ``bash
# -c`` / ``sh -c`` make the call execute a command line through a shell even
# though it carries no ``shell:``/``shell=True`` option (both the JS spawn

_SHELL_BIN = re.compile(r"(?i)\b(sh|bash|zsh|dash)\b")

_SHELL_OPT_TRUE = re.compile(r"shell\s*:\s*true", re.I)

def _first_positional(call, bsrc):
    """``(source_text, tree_sitter_type)`` of the first positional argument,
    or ``None`` for a call with no arguments.  Node type discriminates the
    string-code timer forms: ``arrow_function``/``function`` callbacks are
    safe, while a RHS whose type is not a function *and* whose text contains
    concat/interpolation/external markers is a dynamic code string.
    """
    args = call.child_by_field_name("arguments")
    if args is None:
        return None
    named = [a for a in args.named_children if a.type != "comment"]
    if not named:
        return None
    return _bstr(bsrc, named[0].start_byte, named[0].end_byte), named[0].type


def _first_arg_is_list(call, bsrc) -> bool:
    """True when the first positional argument is a list literal (the argv
    form of subprocess / spawn).  Python parses ``["ls", d]`` with node type
    ``list``; JS/TS parse array literals as ``array``."""
    first = _first_positional(call, bsrc)
    return first is not None and first[1] in ("list", "array")

def _safe_snapshot_path(history_dir: str, name: str) -> Optional[str]:
    """Resolve ``name`` to a snapshot inside ``history_dir`` (safe, or None).

    Accepts an exact snapshot filename or a target slug (newest match, mirroring
    the CLI's ``resolve_snapshot``). Refuses anything that would escape the
    history directory.
    """
    if not name or not re.fullmatch(r"[A-Za-z0-9._-]+", name):
        return None
    hd_real = os.path.realpath(history_dir)
    candidate = os.path.realpath(os.path.join(hd_real, name))
    if not (candidate == hd_real
            or candidate.startswith(hd_real.rstrip(os.sep) + os.sep)):
        return None
    if os.path.isfile(candidate):
        return candidate
    try:
        return resolve_snapshot(name, hd_real)
    except FileNotFoundError:
        return None

def _page_snapshot(history_dir: str, name: str) -> Tuple[int, str, bytes]:
    path = _safe_snapshot_path(history_dir, name)

        .format(score=_e(dim.get("score", 0)),
                cls=_grade_class(doc.get("grade", "?")),
                label=_e(DIMENSION_LABELS.get(d, d)))
        for d, dim in sorted(doc.get("dimensions", {}).items())

    table = _findings_table(doc.get("findings", []))
    suppressed = doc.get("suppressed", [])
    body = (header + '<div class="card"><h2>维度评分</h2>' + dims + "</div>"

            .format(n=len(suppressed), inner=_findings_table(suppressed))

class DashboardHandler(BaseHTTPRequestHandler):
    """Serves the dashboard. ``history_dir`` is injected by ``make_server``."""

    history_dir: str = "."

    def __init__(self, *args, history_dir: Optional[str] = None, **kwargs):
        if history_dir is not None:
            self.history_dir = history_dir
        super().__init__(*args, **kwargs)

    def do_GET(self):
        parsed = urllib.parse.urlsplit(self.path)
        query = dict(urllib.parse.parse_qsl(parsed.query, keep_blank_values=True))
        status, mime, body = route(self.history_dir, parsed.path, query)
        self.send_response(status)
        self.send_header("Content-Type", mime)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("X-Content-Type-Options", "nosniff")
        self.end_headers()
        if self.command != "HEAD":
            self.wfile.write(body)

def make_server(history_dir: str, host: str = "127.0.0.1", port: int = 0):
    """Build a threaded HTTP server for the dashboard (tests pass port=0)."""
    hd = os.path.realpath(history_dir)
    handler = functools.partial(DashboardHandler, history_dir=hd)
    return ThreadingHTTPServer((host, port), handler)


def run_dashboard(history_dir: Optional[str] = None, host: str = "127.0.0.1",
                  port: int = 8080) -> int:
    """Run the dashboard until Ctrl+C. Returns the process exit code."""
    hd = history_dir or default_history_dir()
    os.makedirs(hd, exist_ok=True)
    httpd = make_server(hd, host, port)
    print(f"cursor-scan dashboard: http://{host}:{httpd.server_address[1]}/")
    print(f"history: {os.path.realpath(hd)}   (Ctrl+C 退出)")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        httpd.server_close()
    return 0
